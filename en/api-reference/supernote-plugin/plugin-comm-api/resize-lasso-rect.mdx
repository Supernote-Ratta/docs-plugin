---
title: "resizeLassoRect"
description: "Resize the lasso rectangle."
---
<Note>
You must create a lasso selection before calling this API; otherwise the call fails.
</Note>
```ts
static resizeLassoRect(
  rect: Rect,
  rotationDegree?: number
): Promise<APIResponse<boolean>>;
```

**Parameters**

| Parameter | Type | Description |
| --- | --- | --- |
| `rect` | [`Rect`](/en/api-reference/supernote-plugin/types/rect) | Lasso rectangle (pixel coordinates) |
| `rotationDegree` | `number` | Optional rotation angle in degrees, as an integer. Omit it to keep the current rotation; when provided, supported elements in the current lasso selection rotate by this angle |

**Notes**

- `rotationDegree` applies only to strokes and geometries; other element types are not rotated
- Among lasso data, only titles, strokes, and geometries support stretch resizing; all other element types support proportional scaling only
- Omit `rotationDegree` when rotation is not needed

**Returns**

- [`APIResponse<boolean>`](/en/api-reference/supernote-plugin/types/api-response):
`result === true` indicates the resize succeeded

## Example

```ts wrap
import { PluginCommAPI, type Rect } from 'sn-plugin-lib';

/**
 * Example: resize the lasso rectangle and optionally rotate supported elements.
 */
export async function exampleResizeLassoRect(rotationDegree?: number) {
 const current = await PluginCommAPI.getLassoRect();
 if (!current.success) {
 throw new Error(current.error?.message ?? 'getLassoRect call failed');
 }

 const rect = current.result as Rect;
 const w = rect.right - rect.left;
 const h = rect.bottom - rect.top;
 const cx = rect.left + w / 2;
 const cy = rect.top + h / 2;
 const scale = 1.2;
 const nextRect: Rect = {
 left: cx - (w * scale) / 2,
 top: cy - (h * scale) / 2,
 right: cx + (w * scale) / 2,
 bottom: cy + (h * scale) / 2,
 };

 console.log('resizeLassoRect nextRect:', nextRect, 'rotationDegree:', rotationDegree);
 const res = await PluginCommAPI.resizeLassoRect(nextRect, rotationDegree);
 if (!res.success) {
 throw new Error(res.error?.message ?? 'resizeLassoRect call failed');
 }
 console.log('resizeLassoRect result:', res.result);
 return res.result;
}
```
